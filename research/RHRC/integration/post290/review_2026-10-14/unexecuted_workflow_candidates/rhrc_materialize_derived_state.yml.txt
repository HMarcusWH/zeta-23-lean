name: RHRC materialize derived state for review

on:
  workflow_dispatch:

permissions:
  contents: read

env:
  OPENBLAS_NUM_THREADS: '1'
  OMP_NUM_THREADS: '1'

jobs:
  materialize:
    runs-on: ubuntu-latest
    timeout-minutes: 180
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0
      - uses: actions/setup-python@v5
        with:
          python-version: '3.14'
      - run: python -m pip install -r research/RHRC/requirements.txt
      - uses: actions/cache/restore@v4
        with:
          path: .lake/build
          key: post290-components-${{ hashFiles('lean-toolchain', 'lake-manifest.json') }}-${{ github.sha }}
          restore-keys: post290-components-${{ hashFiles('lean-toolchain', 'lake-manifest.json') }}-
      - uses: leanprover/lean-action@v1
        with:
          auto-config: false
          build: false
          test: false
          lint: false
          use-mathlib-cache: true
      - name: Build the existing compiler exporters
        run: lake build Zeta23.RHRC.RegisteredClaimBindings Zeta23.RHRC.DeclarationDependencyExport Zeta23.RHRC.SourceCandidateResolution
      - name: Materialize the complete deterministic stack
        run: python research/RHRC/tools/materialize_derived_state.py --write
      - name: Independently recheck every materialized product
        run: python research/RHRC/tools/materialize_derived_state.py --check
      - name: Bind the downloadable products to their exact source tree
        run: |
          python - <<'PY'
          import hashlib, json, subprocess
          from pathlib import Path
          roots = [Path('research/RHRC') / p for p in (
              'graph/compiler', 'graph/generated', 'integration/generated',
              'ffbbp/generated', 'ool/generated')]
          files = sorted(p for root in roots for p in root.rglob('*') if p.is_file())
          receipt = {
              'source_head': subprocess.check_output(['git', 'rev-parse', 'HEAD'], text=True).strip(),
              'source_tree': subprocess.check_output(['git', 'rev-parse', 'HEAD^{tree}'], text=True).strip(),
              'generation_and_recheck': 'PASS',
              'fresh_checkout_committed_state_check': 'REQUIRED_AFTER_REVIEW_AND_COMMIT',
              'terminal_claim': 'RH_OPEN',
              'products': {p.as_posix(): hashlib.sha256(p.read_bytes()).hexdigest() for p in files},
          }
          Path('/tmp/materialization-receipt.json').write_text(json.dumps(receipt, indent=2)+'\n')
          PY
      - uses: actions/upload-artifact@v4
        if: success()
        with:
          name: rhkg-materialized-${{ github.sha }}
          path: |
            research/RHRC/graph/compiler/
            research/RHRC/graph/generated/
            research/RHRC/integration/generated/
            research/RHRC/ffbbp/generated/
            research/RHRC/ool/generated/
          if-no-files-found: error
      - uses: actions/upload-artifact@v4
        if: success()
        with:
          name: rhkg-materialization-receipt-${{ github.sha }}
          path: /tmp/materialization-receipt.json
          if-no-files-found: error
