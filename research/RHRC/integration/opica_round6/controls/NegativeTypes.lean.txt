open Zeta23.OPICARound6.PrimeFactorPaths

/-- Deliberately reverse the sign of the off-diagonal product. -/
example : ∀ a b c d : ℝ,
    (a+b+c+d)*(a-b-c+d) - (a+b-c-d)*(a-b+c-d) = 4*(a*d+b*c) :=
  RoughNodeCharges.contingency_covariance

/-- Deliberately drop the unresolved squarefree inequality in the equivalence. -/
example : ∀ x : ℝ, 2 < x → (∫ u in 2..x, (Chebyshev.theta u - u)) < 0 :=
  PrimeAreaIntegral.integral_negative_iff_squarefree_prime_cost

/-- Deliberately replace the opposed-deficit coefficient by five. -/
example (e o : ℝ) :
    0 ≤ (max (e-o) 0)^2 - 5 * max e 0 * max (-o) 0 ∧
    (max (e-o) 0)^2 - 5 * max e 0 * max (-o) 0 ≤ (e+o)^2 :=
  ParityDeficitProduct.quadrant_remainder e o
